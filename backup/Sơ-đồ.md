def measure_sycophancy(text: str) -> float:

def measure_context_noise(a: np.ndarray, b: np.ndarray) -> float:
    norm_a, norm_b = np.linalg.norm(a), np.linalg.norm(b)
    if norm_a < 1e-8 or norm_b < 1e-8:
        return 0.0
    cos = np.dot(a, b) / (norm_a * norm_b)
    return float(max(0.0, 1.0 - cos))

def measure_ontology_gap(projections: List[np.ndarray]) -> float:
    if len(projections) < 2:
        return 0.0
    return max(measure_context_noise(projections[i], projections[j])
               for i in range(len(projections)) for j in range(i+1, len(projections)))

        ancestry_ids = self.trace_lineage_ancestry(node_id, max_depth=max_depth)
        if not ancestry_ids: return 1.0

        entropies = []
        for aid in ancestry_ids:

                    entropies.append(e)
            except: entropies.append(0.11)
        if not entropies: return 0.5
        avg_e = sum(entropies) / len(entropies)
        return float(max(0.0, min(1.0, 1.0 - avg_e)))

    def _compute_reconstruction_score_v4_0171(self, node_id: str, node_data: Dict, query_vec: np.ndarray, query_focus: Dict, context: Optional[Dict] = None) -> float:

        entropy_term = 1.0 - node_data.get('entropy', 0.11)
        attention = node_data.get('attention', 0.72)
        preservation_bonus = 0.0
        if context and 'lineage' in context:
            if node_data.get('lineage', '')!= context.get('lineage', ''):
                preservation_bonus = 0.10
        ancestry_quality = self.compute_ancestry_quality(node_id, max_depth=3)
        score = (0.30 * cos + 0.20 * focus_align + 0.15 * entropy_term + 0.15 * attention + 0.10 * preservation_bonus + 0.10 * ancestry_quality)
        return float(score)
